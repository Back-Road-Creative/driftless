{# One place every server-rendered chart is drawn, so a chart's axes, ticks and legend are written once rather than reinvented per page. A second series is dashed and, when ``rag_class`` is given, tinted by it -- the one place a verdict colours a line, never the plan's own. #} {% macro s_curve(dates, series, aria_label, rag_class=none, w=320, h=140) %} {% set ml, mb, mt = 42, 26, 6 %} {% set plot_w, plot_h = w - ml, h - mb - mt %} {% set n = dates | length %} {% set span = (n - 1) if n > 1 else 1 %} {% set pmax_seen = [1] %} {% for label, values, dashed in series %}{% set _ = pmax_seen.append(values | max) %}{% endfor %} {% set pmax = pmax_seen | max %} {% set ymax = pmax * 1.15 %} {% set ends = [] %} {% for label, values, dashed in series %} {% set _ = ends.append((h - mb - values[-1] / ymax * plot_h) - 4) %} {% endfor %} {% set label_ys = [ends[0]] %} {% if ends | length == 2 %} {% set lo, hi = ([ends[0], ends[1]] | min), ([ends[0], ends[1]] | max) %} {% if hi - lo < 11.2 %} {# Within one label height: stack them.
